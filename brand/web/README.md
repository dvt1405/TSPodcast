# Web icons (for the Next.js site in `web/`)

Generated by `brand/scripts/store.js`. Copy these files into the site's `public/` folder:

| File | Purpose |
|---|---|
| `favicon.ico` | 16 + 32 + 48 px, for legacy browsers and `/favicon.ico` requests |
| `favicon.svg` | Scalable favicon for modern browsers |
| `favicon-16.png`, `favicon-32.png` | PNG fallbacks |
| `apple-touch-icon.png` | 180×180, opaque, full-bleed (iOS rounds it) |
| `icon-192.png`, `icon-512.png` | PWA / Android home-screen icons |
| `icon-maskable-512.png` | PWA maskable icon (content inside the 80% safe zone) |
| `site.webmanifest` | Web app manifest (`lang: vi`, theme `#184BB4`) |
| `og-image.png` | 1200×630 social preview |

## Next.js (App Router) metadata

```ts
// app/layout.tsx
import type { Metadata, Viewport } from 'next';

export const metadata: Metadata = {
  title: { default: 'TSPodcast — Thế giới trong tai bạn', template: '%s · TSPodcast' },
  description: 'Nghe podcast, radio và nhạc miễn phí trên Android.',
  manifest: '/site.webmanifest',
  icons: {
    icon: [
      { url: '/favicon.ico', sizes: 'any' },
      { url: '/favicon.svg', type: 'image/svg+xml' },
      { url: '/favicon-32.png', sizes: '32x32', type: 'image/png' },
      { url: '/favicon-16.png', sizes: '16x16', type: 'image/png' },
    ],
    apple: [{ url: '/apple-touch-icon.png', sizes: '180x180' }],
  },
  openGraph: {
    type: 'website',
    locale: 'vi_VN',
    siteName: 'TSPodcast',
    images: [{ url: '/og-image.png', width: 1200, height: 630, alt: 'TSPodcast — Thế giới trong tai bạn' }],
  },
  twitter: { card: 'summary_large_image', images: ['/og-image.png'] },
};

export const viewport: Viewport = { themeColor: '#184BB4' };
```

Plain HTML equivalent:

```html
<link rel="icon" href="/favicon.ico" sizes="any">
<link rel="icon" href="/favicon.svg" type="image/svg+xml">
<link rel="apple-touch-icon" href="/apple-touch-icon.png">
<link rel="manifest" href="/site.webmanifest">
<meta name="theme-color" content="#184BB4">
<meta property="og:image" content="/og-image.png">
```
